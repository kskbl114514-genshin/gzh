标准 Fetch 模板

// 1. 发起请求
fetch('https://dog.ceo/api/breeds/image/random')
  // 2. 检查响应状态
  .then(function(res) {
    if (!res.ok) throw new Error('HTTP ' + res.status);
    return res.json();   // 3. 解析 JSON
  })
  // 4. 使用数据
  .then(function(data) {
    console.log(data.message);  // 图片 URL
  })
  // 5. 错误处理
  .catch(function(err) {
    console.error(err);
  });
